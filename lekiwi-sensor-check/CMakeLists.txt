cmake_minimum_required(VERSION 3.16)
project(lekiwi_sensor_check LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

add_library(bosch_bmi160 STATIC vendor/bosch_bmi160/bmi160.c)
target_include_directories(bosch_bmi160 PUBLIC vendor/bosch_bmi160)
add_executable(imu_probe native/imu_probe.cpp)
target_link_libraries(imu_probe PRIVATE bosch_bmi160)
target_compile_options(imu_probe PRIVATE -Wall -Wextra -Wpedantic)

set(YDLIDAR_SOURCE "${CMAKE_SOURCE_DIR}/.deps/YDLidar-SDK" CACHE PATH "고정 버전 YDLIDAR SDK 소스")
if(NOT EXISTS "${YDLIDAR_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "YDLIDAR SDK가 없습니다. ./setup.sh를 실행하세요.")
endif()
# SDK의 Python/SWIG 및 ROS 연결 없이 정적 C++ 라이브러리만 빌드한다.
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TEST OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_SWIG TRUE)
set(SWIG_FOUND FALSE)
set(CMAKE_DISABLE_FIND_PACKAGE_PythonInterp TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_PythonLibs TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_GTest TRUE)
set(PYTHONLIBS_FOUND FALSE)
set(GTEST_FOUND FALSE)
add_subdirectory("${YDLIDAR_SOURCE}" "${CMAKE_BINARY_DIR}/ydlidar-sdk" EXCLUDE_FROM_ALL)
add_executable(lidar_probe native/lidar_probe.cpp)
target_include_directories(lidar_probe PRIVATE "${YDLIDAR_SOURCE}" "${YDLIDAR_SOURCE}/src" "${YDLIDAR_SOURCE}/core" "${CMAKE_BINARY_DIR}/ydlidar-sdk")
target_link_libraries(lidar_probe PRIVATE ydlidar_sdk)
target_compile_options(lidar_probe PRIVATE -Wall -Wextra -Wpedantic)
